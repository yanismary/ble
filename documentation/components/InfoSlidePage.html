import { OnInit, Component } from '@angular/core';
import { IonicPage, NavController, NavParams } from 'ionic-angular';
import { TranslateService } from '@ngx-translate/core';


/**
 * Generated class for the InfoSlidePage page.
 *
 * See https://ionicframework.com/docs/components/#navigation for more info on
 * Ionic pages and navigation.
 */

@IonicPage()
@Component({
  selector: 'page-infoSlide',
  templateUrl: 'infoSlide.html',
})
export class InfoSlidePage implements OnInit {
  slides: any[] = [];

  constructor(
    public navCtrl: NavController,
    public navParams: NavParams,
    private translate: TranslateService,


  ) {


  }


  ngOnInit(): any {


    this.translate.get(['INFOSLIDE.SLIDE1.TITLE','INFOSLIDE.SLIDE1.DESC']).subscribe(
      res => {

        this.slides = [
          {
            title: res["INFOSLIDE.SLIDE1.TITLE"],
            description: res["INFOSLIDE.SLIDE1.DESC"],
            image: "./assets/img/slide1.png"
          },   
          {
            title: res["INFOSLIDE.SLIDE2.TITLE"],
            description: res["INFOSLIDE.SLIDE2.DESC"],
            image: "./assets/img/slide2.png"
          },     
          {
            title: "How pair with Moventiv ?",
            description: "On the phone side go to the parameters-> connections->Bluetooth-> search devices. On the Moventiv side, put the Switch 1 on OFF state, press the prg button (LED indicator should be magenta). Then on the phone side, select the Moventiv Device. It will be put on the paired devices",
            image: "./assets/img/slide2.png"
          },
          {
            title: "Rename your device",
            description: "You can rename your Moventiv in the tuning menu and you can associate it with a room",
            image: "assets/img/ica-slidebox-img-3.png",
          }
        ];
      });





  }

  ionViewDidEnter() {
    //cosmetic
  }





  pushScan() {
    this.navCtrl.push('ScanPage');
  }





  ionViewDidLoad() {
    console.log('ionViewDidLoad InfoSlidePage');
  }

}
