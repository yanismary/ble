  scan() {

    this.setStatus('Scanning for MLPC Device');

    this.isScanning = true;
    this.isPushOnce = false;

    let scanParams = {
      allowDuplicates: false,
      matchNum: this.randble.MATCH_NUM_MAX_ADVERTISEMENT,
      callbackType: this.randble.CALLBACK_TYPE_ALL_MATCHES,
      scanMode: this.randble.SCAN_MODE_BALANCED, //more efficient scan for crowed place (the LE mode seems not efficient for common use)
      services: [WIDOOR_SERVICE],
    };

    this.devices = [];  // clear list

    this.randble.startScan(scanParams).subscribe(
      device => (this.onDeviceDiscovered(device)
        // this.device=device
      ),

    );
    if (this.isScanning) {
      setTimeout(() => {
        if (this.isScanning)
          this.randble.stopScan().then(
            () => {
              console.log("Scanning has stopped");
              this.ngZone.run(() => {
                this.isScanning = false;
              });
            },
            () => {
              console.log("Error at stop scan");
              this.ngZone.run(() => {
                this.isScanning = false;
              });
            }
          );

      }, 6000);
    }
  }

  onDeviceDiscovered(device) {
    console.log('Enter in OnDeviceDiscovered1');
    if ((device.status) == "scanStarted") {//if scanStarted all other fields are undefined so just return
      console.log('Scan started');
      return
    }

    //is the device already bond ? Android only
    //if(0){
    if (this.platform.is('android')) {
      this.randble.isBonded({ address: device.address }).subscribe(
        deviceBond => {
          this.onDeviceDiscovered2(Object.assign(device, deviceBond));
          console.log('isBonded : ' + device.isBonded);
        },
        () => {
          console.log('isBonded error');
        }
      );


    }
    else {
      this.onDeviceDiscovered2(device);
    }

  }


  onDeviceDiscovered2(device) {
    console.log('Enter in OnDeviceDiscovered2');
    var alreadyExist = false;
    this.ngZone.run(() => {
      console.log('Discovered ' + JSON.stringify(device, null, 2));
      for (var i = 0; i < this.devices.length; i++) {

        //the goal of the following part is to have the RSSI refreshing during the discover
        if ((this.devices[i].address) == (device.address)) { // check if already exist
          alreadyExist = true;
          this.devices.splice(i, 1, device); //delete the old one, push the "new" at the same place in the array    
        }
      }

      if (alreadyExist == false) { // if not already exist push it into array 
        this.devices.push(device);
      }

    });

  }
