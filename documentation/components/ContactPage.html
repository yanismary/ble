import { Component, ViewChild } from '@angular/core';
import { NavController, NavParams, Platform, Navbar } from 'ionic-angular';
import { EmailComposer } from '@ionic-native/email-composer';
import { IonicPage } from 'ionic-angular';
import { TranslateService } from '@ngx-translate/core';
import { CallNumber } from '@ionic-native/call-number';

/**
 * Generated class for the ContactPage page.
 *
 * See http://ionicframework.com/docs/components/#navigation for more info
 * on Ionic pages and navigation.
 */

@IonicPage()
@Component({
  selector: 'page-contact',
  templateUrl: 'contact.html',
})
export class ContactPage {

  @ViewChild(Navbar) navBar: Navbar;
  //translate string
  translate_cp_mc_to: string;
  translate_cp_mc_cc: string;
  translate_cp_mc_s: string;
  translate_cp_mc_body: string;
  translate_cp_msc_to: string;
  translate_cp_msc_cc: string;
  translate_cp_msc_s: string;
  translate_cp_msc_body: string;

  fromConnected: boolean;

  constructor(


    public navCtrl: NavController,
    public navParams: NavParams,
    private emailComposer: EmailComposer,
    private translate: TranslateService,
    public callNumber: CallNumber,
    public platform: Platform,
  ) {

    this.translate.get('CONTACT_PAGE.MANTION_CONTACT.SEND_MESSAGE.TO').subscribe(res => {
      this.translate_cp_mc_to = res;
    });
    this.translate.get('CONTACT_PAGE.MANTION_CONTACT.SEND_MESSAGE.CC').subscribe(res => {
      this.translate_cp_mc_cc = res;
    });
    this.translate.get('CONTACT_PAGE.MANTION_CONTACT.SEND_MESSAGE.SUBJECT').subscribe(res => {
      this.translate_cp_mc_s = res;
    });
    this.translate.get('CONTACT_PAGE.MANTION_CONTACT.SEND_MESSAGE.BODY').subscribe(res => {
      this.translate_cp_mc_body = res;
    });
    this.translate.get('CONTACT_PAGE.MANTIONSMT_CONTACT.SEND_MESSAGE.TO').subscribe(res => {
      this.translate_cp_msc_to = res;
    });
    this.translate.get('CONTACT_PAGE.MANTIONSMT_CONTACT.SEND_MESSAGE.CC').subscribe(res => {
      this.translate_cp_msc_cc = res;
    });
    this.translate.get('CONTACT_PAGE.MANTIONSMT_CONTACT.SEND_MESSAGE.SUBJECT').subscribe(res => {
      this.translate_cp_msc_s = res;
    });
    this.translate.get('CONTACT_PAGE.MANTIONSMT_CONTACT.SEND_MESSAGE.BODY').subscribe(res => {
      this.translate_cp_msc_body = res;
    });

    this.fromConnected = navParams.get('fromConnected');
  }


  sendMessageType1() {
    let email = {
      to: this.translate_cp_mc_to,
      cc: this.translate_cp_mc_cc,

      attachments: [

      ],
      subject: this.translate_cp_mc_s,
      body: this.translate_cp_mc_body,
      isHtml: true
    };

    this.emailComposer.open(email);

  }

  sendMessageType2() {
    let email = {
      to: this.translate_cp_msc_to,
      cc: this.translate_cp_msc_cc,

      attachments: [

      ],
      subject: this.translate_cp_msc_s,
      body: this.translate_cp_msc_body,
      isHtml: true
    };

    this.emailComposer.open(email);

  }

  callPhoneNumberM() {
    this.callNumber.callNumber("33381505677", false)
      .then(() => console.log('Launched dialer!'))
      .catch(() => console.log('Error launching dialer'));
  }

  callPhoneNumberMSMT() {
    this.callNumber.callNumber("33380378571", false)
      .then(() => console.log('Launched dialer!'))
      .catch(() => console.log('Error launching dialer'));
  }

  //lifecycle

  ionViewDidLoad() {
    console.log('ionViewDidLoad ContactPage');
  }

  ionViewWillEnter() {
    this.platform.registerBackButtonAction(() => this.setBackButtonActionHW());
    this.setBackButtonActionSW();
    console.log('ionViewWillEnter');
  }

  ionViewDidEnter() {
    //cosmetic

  }

  ionViewCanLeave() {

    //this.navCtrl.popToRoot();

  }

  //Method to override the default back button action
  private setBackButtonActionSW() {
    this.navBar.backButtonClick = () => {
      //Write here wherever you wanna do
      console.log('backButtonFunc()');
      this.navCtrl.pop();
    }
  }

  private setBackButtonActionHW() {

    this.navCtrl.pop();
  }


}
